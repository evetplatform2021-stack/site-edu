12.2. Изграждане на Async Core Event Bus

Шината за събития (Event Bus) осигурява пълен Декуплинг (Decoupling) между източниците на данни и консуматорите. Използваме typing.Generic и `weakref` за предотвратяване на циклично задържане на памет от обработчиците (Handler leaks).

event_bus.py
import asyncio
import weakref
from typing import TypeVar, Generic, Callable, Awaitable, Dict, List
from dataclasses import dataclass

T = TypeVar("T")

@dataclass(frozen=True, slots=True)
class BaseEvent:
    timestamp: float

EventHandler = Callable[[T], Awaitable[None]]

class EventBus(Generic[T]):
    def __init__(self) -> None:
        self._subscribers: Dict[type, List[weakref.WeakMethod]] = {}

    def subscribe(self, event_type: type, handler: EventHandler[T]) -> None:
        if event_type not in self._subscribers:
            self._subscribers[event_type] = []
        # Използва се WeakMethod за избягване на Memory Leaks при регистриране на методи
        self._subscribers[event_type].append(weakref.WeakMethod(handler))

    async def publish(self, event: T) -> None:
        event_type = type(event)
        if event_type not in self._subscribers:
            return

        active_subscribers = []
        tasks = []

        for ref in self._subscribers[event_type]:
            func = ref()
            if func is not None:
                active_subscribers.append(ref)
                tasks.append(func(event))

        # Премахване на мъртви референции (Garbage Collection Cleanup)
        self._subscribers[event_type] = active_subscribers

        if tasks:
            await asyncio.gather(*tasks, return_exceptions=True)

12.3. High-Performance Hotpath с CPython Optimization

Критичната обработка на числови агрегати в телеметрията се изпълнява през векторизирана нулева алокация, заобикаляйки типичния overhead на Python обектите.

hotpath_processor.py
import numpy as np

class BatchTelemetryAggregator:
    __slots__ = ("_buffer", "_capacity", "_cursor")

    def __init__(self, capacity: int = 10_000):
        self._capacity = capacity
        # Pre-allocated C-array за избягване на регулярни allocations
        self._buffer = np.zeros(capacity, dtype=np.float64)
        self._cursor = 0

    def push(self, val: float) -> bool:
        if self._cursor >= self._capacity:
            return False  # Буферът е пълен
        self._buffer[self._cursor] = val
        self._cursor += 1
        return True

    def flush_stats(self) -> dict:
        if self._cursor == 0:
            return {"mean": 0.0, "p99": 0.0}

        view = self._buffer[:self._cursor]
        stats = {
            "mean": float(np.mean(view)),
            "p99": float(np.percentile(view, 99)),
            "count": self._cursor
        }
        self._cursor = 0  # Fast Reset без освобождаване на памет
        return stats

12.4. Concurrency, Rate Limiting & Backpressure Mechanism

Когато входният поток надвишава скоростта на обработка, системата трябва да приложи Backpressure чрез оразмерени опашки (`bounded queue`), вместо да позволява неограничено разширение на RAM.

backpressure_pipeline.py
import asyncio
import logging

class BoundedPipeline:
    def __init__(self, max_queue_size: int = 1000, max_workers: int = 4):
        self.queue: asyncio.Queue = asyncio.Queue(maxsize=max_queue_size)
        self.max_workers = max_workers
        self.workers: list[asyncio.Task] = []

    async def produce(self, item: dict) -> bool:
        try:
            # Неблокиращо пускане; хвърля Full ако опашката е препълнена (Backpressure trigger)
            self.queue.put_nowait(item)
            return True
        except asyncio.QueueFull:
            logging.warning("Backpressure Triggered! Dropping or throttling incoming event.")
            return False

    async def _worker_loop(self, worker_id: int):
        while True:
            item = await self.queue.get()
            try:
                # Симулация на обработка
                await asyncio.sleep(0.001)
            finally:
                self.queue.task_done()

    async def start(self):
        for i in range(self.max_workers):
            self.workers.append(asyncio.create_task(self._worker_loop(i)))

    async def stop(self):
        await self.queue.join()
        for t in self.workers:
            t.cancel()

12.6. Телеметрия и Мониторинг (Structured Logging & Metrics)

Използване на структурирани JSON логове без блокиращи фаза-I/O операции, подходящи за Grafana / Kibana инжектиране.

telemetry.py
import json
import time
import logging

class JSONFormatter(logging.Formatter):
    def format(self, record: logging.LogRecord) -> str:
        log_obj = {
            "ts": time.time(),
            "level": record.levelname,
            "message": record.getMessage(),
            "module": record.module
        }
        if hasattr(record, "extra_data"):
            log_obj["extra"] = record.extra_data
        return json.dumps(log_obj)

handler = logging.StreamHandler()
handler.setFormatter(JSONFormatter())
logger = logging.getLogger("EngineLogger")
logger.addHandler(handler)
logger.setLevel(logging.INFO)

12.7. Пълен Изходен Код на Capstone Двигателя

Сглобяване на всички компоненти в единен работен скрипт с чист Graceful Shutdown при прекъсване (SIGINT/SIGTERM).

main_engine.py
import asyncio
import time
import signal
from event_bus import EventBus, BaseEvent
from hotpath_processor import BatchTelemetryAggregator
from backpressure_pipeline import BoundedPipeline

class SystemMetricEvent(BaseEvent):
    def __init__(self, val: float):
        super().__init__(timestamp=time.time())
        self.val = val

async def main():
    bus = EventBus[SystemMetricEvent]()
    aggregator = BatchTelemetryAggregator(capacity=5000)
    pipeline = BoundedPipeline(max_queue_size=500, max_workers=2)

    await pipeline.start()

    async def handle_metric(evt: SystemMetricEvent):
        success = aggregator.push(evt.val)
        if not success:
            stats = aggregator.flush_stats()
            print(f"[Flush Stats] {stats}")
            aggregator.push(evt.val)

    bus.subscribe(SystemMetricEvent, handle_metric)

    print("🚀 Event Engine стартиран успешно. Натиснете Ctrl+C за спиране...")

    try:
        for i in range(12_000):
            event = SystemMetricEvent(val=float(i % 100))
            await bus.publish(event)
            if i % 1000 == 0:
                await asyncio.sleep(0.01) # Даваме глътка въздух на Event Loop-а
    except asyncio.CancelledError:
        pass
    finally:
        await pipeline.stop()
        final_stats = aggregator.flush_stats()
        print(f"🛑 Engine спрян успешно. Финални статистики: {final_stats}")

if __name__ == "__main__":
    try:
        asyncio.run(main())
    except KeyboardInterrupt:
        print("\nПрограмата е прекратена от потребителя.")
